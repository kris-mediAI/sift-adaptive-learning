def current_session():
    if st.session_state.learner_id is None:
        return None
    return sift.get_session(st.session_state.learner_id)


def current_learner():
    session = current_session()
    return session.learner if session else None


def current_concepts():
    session = current_session()
    return list(session.concepts.values()) if session else []


def get_recommendation():
    session = current_session()
    if not session:
        return None
    return session.engine.recommend(
        learner=session.learner,
        concepts=list(session.concepts.values()),
    )


def safe_float(value, default=0.0):
    try:
        return float(value)
    except (TypeError, ValueError):
        return default


def mastery_average(concepts):
    if not concepts:
        return 0.0
    return sum(safe_float(c.mastery) for c in concepts) / len(concepts)


def latest_gain(learner):
    records = getattr(learner, "learning_records", [])
    if not records:
        return 0.0
    gain = getattr(records[-1], "learning_gain", 0)
    return safe_float(gain)


def strategy_name(strategy):
    return str(strategy or "").replace("_", " ")

def _get_value(obj, name, default=None):
    """
    Safely read an attribute from an object or key from a dict.
    """

    if obj is None:
        return default

    if isinstance(obj, dict):
        return obj.get(name, default)

    return getattr(obj, name, default)


def _get_concepts(learner):
    """
    Safely extract concept history from the learner.
    """

    history = _get_value(
        learner,
        "concept_history",
        {},
    )

    if not isinstance(history, dict):
        return {}

    return history


def _get_overall_mastery(learner):
    """
    Calculate a lightweight overall mastery value.

    This is UI-only aggregation. It does not modify
    learner state or call the adaptive engine.
    """

    concepts = _get_concepts(learner)

    values = []

    for data in concepts.values():

        if not isinstance(data, dict):
            continue

        mastery = data.get(
            "current_mastery"
        )

        if mastery is None:
            mastery = data.get(
                "mastery"
            )

        try:
            values.append(
                float(mastery)
            )
        except (
            TypeError,
            ValueError,
        ):
            continue

    if not values:
        return 0.0

    return sum(values) / len(values)


def _get_focus(
    recommendation,
    learner,
):
    """
    Convert the adaptive recommendation into
    presentation-friendly data.

    No recommendation is generated here.
    """

    if not isinstance(
        recommendation,
        dict,
    ):
        recommendation = {}

    concept = (
        recommendation.get(
            "concept"
        )
        or recommendation.get(
            "target_concept"
        )
    )

    if not concept:

        concepts = _get_concepts(
            learner
        )

        if concepts:
            concept = next(
                iter(concepts)
            )

    if not concept:
        concept = "Your next concept"

    action = (
        recommendation.get(
            "action"
        )
        or "practice"
    )

    strategy = (
        recommendation.get(
            "strategy"
        )
        or "adaptive"
    )

    reason = (
        recommendation.get(
            "reason"
        )
    )

    if reason:
        description = reason
    else:
        description = (
            "Sift has selected a focused "
            f"{action} session using a "
            f"{strategy.replace('_', ' ')} approach."
        )

    return (
        concept,
        description,
        strategy,
    )
